# Vulkan transcriptions of libScePsm's embedded shaders -> bin/shaders/<name>.spv
file(GLOB SSN_PSM_SHADERS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/shaders/psm/*.vert" "${CMAKE_CURRENT_SOURCE_DIR}/shaders/psm/*.frag"
  "${CMAKE_CURRENT_SOURCE_DIR}/shaders/bglayer/*.vert" "${CMAKE_CURRENT_SOURCE_DIR}/shaders/bglayer/*.frag")
set(SSN_SPV_OUTPUTS)
foreach(src ${SSN_PSM_SHADERS})
  get_filename_component(stem "${src}" NAME_WE)
  set(spv "${CMAKE_BINARY_DIR}/bin/shaders/${stem}.spv")
  add_custom_command(OUTPUT "${spv}"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/bin/shaders"
    COMMAND Vulkan::glslangValidator -V --quiet -o "${spv}" "${src}"
    DEPENDS "${src}" COMMENT "glslang ${stem}")
  list(APPEND SSN_SPV_OUTPUTS "${spv}")
endforeach()
# Background-layer particle and transition shaders: generated by tools/shaders/isa2glsl.py from the
# eboot's ISA. They use subgroup votes (the wave-uniform branches of the original), hence SPIR-V 1.6.
file(GLOB SSN_PARTICLE_SHADERS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/shaders/bglayer/particle/*.comp"
  "${CMAKE_CURRENT_SOURCE_DIR}/shaders/bglayer/particle/*.vert" "${CMAKE_CURRENT_SOURCE_DIR}/shaders/bglayer/particle/*.frag"
  "${CMAKE_CURRENT_SOURCE_DIR}/shaders/bglayer/transition/*.frag")
foreach(src ${SSN_PARTICLE_SHADERS})
  get_filename_component(stem "${src}" NAME_WE)
  set(spv "${CMAKE_BINARY_DIR}/bin/shaders/bglayer_${stem}.spv")
  add_custom_command(OUTPUT "${spv}"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/bin/shaders"
    COMMAND Vulkan::glslangValidator -V --quiet --target-env vulkan1.3 -o "${spv}" "${src}"
    DEPENDS "${src}" COMMENT "glslang ${stem}")
  list(APPEND SSN_SPV_OUTPUTS "${spv}")
endforeach()
# libScePsm GPU texture compressor shaders: generated by tools/shaders/isa2glsl.py from libScePsm's ISA.
file(GLOB SSN_GTC_SHADERS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/shaders/psm/gtc/*.comp")

add_custom_target(psm_shaders ALL DEPENDS ${SSN_SPV_OUTPUTS})
